External microservices, hedge funds, and enterprise trading desks can install and consume `kronos-apex-financial-os` directly from npm:

### 1. Installation
```bash

```

### 2. Programmatic TypeScript / Node.js SDK
```typescript
import { 
  KronosEngine, 
  EpicStoriesEngine, 
  QboQuantumBridge, 
  DEFAULT_BANK_FEEDS 
} from 'kronos-apex-financial-os';

// Initialize the Sovereign Financial Kernel
const kronos = new KronosEngine({
  apiKey: process.env.GEMINI_API_KEY,
  environment: 'sandbox', // or 'production'
  realmId: '9341453267972001',
  port: 3000
});

async function main() {
  console.log('[KRONOS] Executing automated multi-bank settlement cycle...');

  // Ingest Citi Institutional Global Wire Feed
  const feed = DEFAULT_BANK_FEEDS[0];

  // 1. Synthesize Dual Forensic Audit & Epic Story
  const auditReport = await kronos.epicStories.generateReport({
    mode: 'all', // 'professional' | 'narrative' | 'all'
    feedId: feed.id,
    borderStyle: 'gilded-gold',
    customPrompt: 'Analyze liquidity impact of $4.25M Alpha Centauri settlement.'
  });

  console.log(auditReport.note);

  // 2. Double-Entry Auto-Reconciliation into QuickBooks Chart of Accounts
  const result = await kronos.bridge.reconcileLedger({
    autoClassify: true,
    isoStandards: ['pacs.008', 'camt.053']
  });

  console.log(`Reconciled ${result.reconciledCount} items. Status: ${result.status}`);
}

main();
```
